\usepackage{tikz}
\usetikzlibrary{arrows.meta, positioning, fit, backgrounds, calc}
\begin{document}
\begin{tikzpicture}[
  font=\small,
  >={Stealth[length=2.6mm]},
  phase/.style={
    rectangle, rounded corners=3pt, draw=black, line width=0.6pt,
    fill=black!5, minimum height=13mm, text width=27mm, align=center,
    inner sep=3pt},
  io/.style={
    rectangle, rounded corners=1pt, draw=black, line width=0.6pt,
    fill=white, minimum height=13mm, text width=18mm, align=center,
    inner sep=3pt},
  example/.style={
    rectangle, rounded corners=2pt, draw=black!55, line width=0.45pt,
    fill=black!2, minimum height=14mm, text width=27mm, align=center,
    inner sep=2.5pt, font=\scriptsize},
  lbl/.style={font=\scriptsize\itshape, text=black!70},
  flow/.style={->, line width=0.9pt},
]

% Input
\node[io] (query) {Natural-language query};

% Four phases
\node[phase, right=6mm of query] (disc)
  {\textbf{Discovery}\\[1pt]\scriptsize web + PDF search, LLM query expansion};
\node[phase, right=6mm of disc] (extr)
  {\textbf{Extraction}\\[1pt]\scriptsize LLM $\rightarrow$ JSON; regex fallback};
\node[phase, right=6mm of extr] (synth)
  {\textbf{Synthesis}\\[1pt]\scriptsize copula rank transform; mixed types};
\node[phase, right=6mm of synth] (valid)
  {\textbf{Validation}\\[1pt]\scriptsize KS + corr.\ + bias; row dispersion};

% Output
\node[io, right=6mm of valid] (out)
  {Synthetic dataset + data card};

% Main flow arrows
\draw[flow] (query) -- (disc);
\draw[flow] (disc) -- (extr);
\draw[flow] (extr) -- (synth);
\draw[flow] (synth) -- (valid);
\draw[flow] (valid) -- (out);

% Phase band label
\begin{scope}[on background layer]
  \node[fit=(disc)(valid), draw=black!40, dashed, rounded corners=4pt,
        inner sep=6mm, label={[lbl, yshift=-1mm]above:Assisted agentic pipeline}]
        (band) {};
\end{scope}

% Feedback loop: Validation -> Synthesis (data-free self-refinement)
\draw[flow, black!65]
  (valid.north) .. controls +(0,9mm) and +(0,9mm) .. (synth.north)
  node[midway, above, lbl, yshift=0.5mm]
  {noise-pivot self-refinement (ratchet, no real data)};

% Concrete, source-backed walkthrough. The omission is shown deliberately:
% intrinsic fidelity cannot detect a statistic the extractor failed to return.
\node[example, below=13mm of query] (qex)
  {\textbf{Example query}\\smartphone/internet addiction and mental health};
\node[example, below=13mm of disc] (dex)
  {\textbf{Open PMC paper}\\rank 1; total sample $N{=}447$};
\node[example, below=13mm of extr] (eex)
  {\textbf{LLM output}\\SAS-SV $35.7{\pm}8.9$; source $r{=}.64$ omitted};
\node[example, below=13mm of synth] (sex)
  {\textbf{Generated}\\447 mixed-type rows from returned summaries};
\node[example, below=13mm of valid] (vex)
  {\textbf{Intrinsic check}\\96.81; omission remains invisible};
\node[example, below=13mm of out] (oex)
  {\textbf{Artifacts}\\CSV + scoped data card};

\foreach \a/\b in {query/qex,disc/dex,extr/eex,synth/sex,valid/vex,out/oex}
  \draw[densely dashed, black!45] (\a.south) -- (\b.north);
\draw[flow, black!55] (qex) -- (dex);
\draw[flow, black!55] (dex) -- (eex);
\draw[flow, black!55] (eex) -- (sex);
\draw[flow, black!55] (sex) -- (vex);
\draw[flow, black!55] (vex) -- (oex);

\end{tikzpicture}
